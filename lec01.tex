\section{Numbers}

\begin{definition}[Natural Numbers]{natural-number}
The \emph{natural numbers} are the positive integers:
\[
  \mathbb{N}=\{1,2,3,\ldots\}.
\]
In these notes, \(0\notin\mathbb{N}\).
\end{definition}

\begin{definition}[Integers]{integer}
The \emph{integers} are the whole numbers, including the positive integers,
the negative integers, and zero:
\[
  \mathbb{Z}=\{\ldots,-2,-1,0,1,2,\ldots\}.
\]
\end{definition}

% With no explicit ID, this automatically creates def:rational-numbers.
\begin{definition}[Rational Numbers]
A real number is called \emph{rational} if it can be expressed as a quotient
of two integers. The set of rational numbers is
\[
  \mathbb{Q}
  =\left\{\frac{p}{q}:p,q\in\mathbb{Z},\ q\neq0\right\}.
\]
\end{definition}

\begin{remark}{number-set-inclusions}
Under the usual identifications,
\(\mathbb{N}\subsetneq\mathbb{Z}\subsetneq\mathbb{Q}\).
\end{remark}

\section{Fields}

\begin{definition}[Closed Operation]
An operation $\ast$ is said to be \emph{closed} on a set $A$ if for all $a,b \in A$, we have $a \ast b \in A$. 
\end{definition}

\begin{definition}[Field]{field}
A \emph{field} is a set of numbers or elements where addition and
multiplication are \xref[closed operations]{def:closed-operation}.
\end{definition}
