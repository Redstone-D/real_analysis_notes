\section{Intervals}

\begin{dfn}[Interval]{interval}
  A set \(I\subseteq\mathbb{R}\) is an \emph{interval} if, whenever
  \(x,y\in I\) with \(x\le y\), every \(z\in\mathbb{R}\) satisfying
  \(x\le z\le y\) also belongs to \(I\).
\end{dfn}

\begin{rem}[Intervals as Sets]{intervals-as-sets}
  In these notes, every interval is a set contained in \(\mathbb{R}\);
  that is, every interval \(I\) satisfies \(I\subseteq\mathbb{R}\).
\end{rem}

\begin{dfn}[Open Interval]{open-interval}
  For \(a,b\in\mathbb{R}\) with \(a<b\), the \emph{open interval}
  from \(a\) to \(b\) is
  \[
    (a,b)=\{x\in\mathbb{R}:a<x<b\}.
  \]
\end{dfn}

\begin{dfn}[Closed Interval]{closed-interval}
  For \(a,b\in\mathbb{R}\) with \(a\le b\), the \emph{closed interval}
  from \(a\) to \(b\) is
  \[
    [a,b]=\{x\in\mathbb{R}:a\le x\le b\}.
  \]
\end{dfn}
