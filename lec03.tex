\section{Metric Spaces}

\begin{dfn}[Metric]{metric}
  Let \(X\) be a set. A \emph{metric}, or \emph{distance function}, on
  \(X\) is a function \(d:X\times X\to\RR\) satisfying the following
  four axioms:
  \xref[nonnegativity]{axm:metric-nonnegativity},
  \xref[identity of indiscernibles]{axm:metric-separation},
  \xref[symmetry]{axm:metric-symmetry}, and
  \xref[the triangle inequality]{axm:metric-triangle-inequality}
\end{dfn}

\begin{axm}[Nonnegativity]{metric-nonnegativity}
  If \(d\) is a metric on \(X\), then
  \[
    \forall x,y\in X,\quad d(x,y)\geq0
  \]
\end{axm}

\begin{axm}[Identity of Indiscernibles]{metric-separation}
  If \(d\) is a metric on \(X\), then
  \[
    \forall x,y\in X,\quad d(x,y)=0\iff x=y
  \]
\end{axm}

\begin{axm}[Symmetry]{metric-symmetry}
  If \(d\) is a metric on \(X\), then
  \[
    \forall x,y\in X,\quad d(x,y)=d(y,x)
  \]
\end{axm}

\begin{axm}[Triangle Inequality]{metric-triangle-inequality}
  If \(d\) is a metric on \(X\), then
  \[
    \forall x,y,z\in X,\quad d(x,z)\leq d(x,y)+d(y,z)
  \]
\end{axm}

\begin{dfn}[Metric Space]{metric-space}
  A \emph{metric space} is a pair \((X,d)\), where \(X\) is a set and
  \(d\) is a \xref[metric]{def:metric} on \(X\)
\end{dfn}

\section{Open Balls and Neighbourhoods}

\begin{dfn}[Open Balls]{open-balls}
  Let \((X,d)\) be a \xref[metric space]{def:metric-space}.
  The \emph{open-ball generator} is
  \[
    \OpenBall:\apos{\RR}\to\bigl(X\to\PowerSet(X)\bigr)
    \defeq r\map x\map\{y\in X:d(x,y)<r\}
  \]
  For \(r\in\apos{\RR}\) and \(x\in X\),
  \(\OpenBall(r)(x)\) is the \emph{open ball centred at \(x\)
  with radius \(r\)}
  \par\smallskip\noindent{\footnotesize\texttt{LaTeX: \detokenize{\OpenBall(r)(x)}}}
\end{dfn}

\begin{rem}[Open Ball Intuition]{open-ball-intuition}
  An \xref[open ball]{def:open-balls} in a
  \xref[metric space]{def:metric-space} \((X,d)\), centred at
  \(x\in X\) with positive radius \(r\), is the collection of all
  elements of \(X\) whose distance from \(x\) is less than \(r\)
\end{rem}

\begin{conv}[Open Ball Notation]{open-ball-notation}
  For the \xref[open-ball generator]{def:open-balls}, write
  \[
    \ball{r}{x}\defeq\OpenBall(r)(x)
  \]
  \par\smallskip\noindent{\footnotesize\texttt{LaTeX: \detokenize{\ball{r}{x}}}}
\end{conv}

\begin{dfn}[Neighbourhood]{neighbourhood}
  Let \((X,d)\) be a \xref[metric space]{def:metric-space}, and let
  \(x\in X\). A set \(U\subseteq X\) is a \emph{neighbourhood of \(x\)} if
  \[
    \exists r\in\apos{\RR},\quad\ball{r}{x}\subseteq U
  \]
\end{dfn}

\begin{dfn}[Bounded Set in a Metric Space]{metric-bounded-set}
  Let \((X,d)\) be a \xref[metric space]{def:metric-space}.
  A subset \(S\subseteq X\) is \emph{bounded} if
  \[
    \exists M\in\apos{\RR},\,\forall x,y\in S,\quad d(x,y)<M
  \]
\end{dfn}

\begin{conv}[Radius Neighbourhood Notation]{radius-neighbourhood-notation}
  In a \xref[metric space]{def:metric-space} \((X,d)\), some texts call
  the \xref[open ball]{def:open-balls} \(\ball{r}{x}\), with
  \(r\in\apos{\RR}\) and \(x\in X\), the
  \emph{\(r\)-neighbourhood of \(x\)} and write
  \[
    V_r(x)\defeq\ball{r}{x}
  \]
\end{conv}

\begin{lem}[The Empty Set and the Whole Space]{metric-empty-and-whole-open}
  Let \((X,d)\) be a \xref[metric space]{def:metric-space}. Then
  \[
    \forall U\in\{\emptyset,X\},\,\forall x\in U,\,
    \exists r\in\apos{\RR},\quad
    \ball{r}{x}\subseteq U
  \]
\end{lem}

\begin{proof}
  \noindent\textbf{Case \(U=X\):} For every \(x\in X\), choose
  \(r=1\). By definition, \(\ball{1}{x}\subseteq X\)

  \noindent\textbf{Case \(U=\emptyset\):} There is no
  \(x\in\emptyset\), so the universally quantified condition holds
  vacuously

  The empty-set case does not require finding a ball contained in
  \(\emptyset\), since there is no centre to consider
\end{proof}

\begin{lem}[Finite Intersections]{metric-finite-intersections}
  Let \((X,d)\) be a \xref[metric space]{def:metric-space}, and let
  \(U,V\subseteq X\) satisfy
  \[
    \begin{aligned}
      &\forall x\in U,\,\exists r\in\apos{\RR},\quad
        \ball{r}{x}\subseteq U\\
      &\forall x\in V,\,\exists r\in\apos{\RR},\quad
        \ball{r}{x}\subseteq V
    \end{aligned}
  \]
  Then
  \[
    \forall x\in U\cap V,\,\exists r\in\apos{\RR},\quad
    \ball{r}{x}\subseteq U\cap V
  \]
\end{lem}

\begin{proof}
  Let \(x\in U\cap V\). By the assumptions, choose
  \(r_U,r_V\in\apos{\RR}\) such that
  \(\ball{r_U}{x}\subseteq U\) and \(\ball{r_V}{x}\subseteq V\).
  Let \(r=\min(r_U,r_V)>0\). Then
  \[
    \ball{r}{x}\subseteq\ball{r_U}{x}\subseteq U,
    \qquad
    \ball{r}{x}\subseteq\ball{r_V}{x}\subseteq V
  \]
  Therefore, \(\ball{r}{x}\subseteq U\cap V\)
\end{proof}

\begin{lem}[Arbitrary Unions]{metric-arbitrary-unions}
  Let \((X,d)\) be a \xref[metric space]{def:metric-space}, and let
  \(U:I\to\PowerSet(X)\) satisfy
  \[
    \forall i\in I,\,\forall x\in U(i),\,
    \exists r\in\apos{\RR},\quad
    \ball{r}{x}\subseteq U(i)
  \]
  Then
  \[
    \forall x\in\bigcup_I U,\,\exists r\in\apos{\RR},\quad
    \ball{r}{x}\subseteq\bigcup_I U
  \]
\end{lem}

\begin{proof}
  Let \(x\in\bigcup_I U\). Choose \(i\in I\) with \(x\in U(i)\).
  By the assumption, choose \(r\in\apos{\RR}\) such that
  \[
    \ball{r}{x}\subseteq U(i)\subseteq\bigcup_I U
  \]
\end{proof}

\section{Distance on an Ordered Field}

\begin{dfn}[Distance]{distance}
  Let \(F\) be an \xref[ordered field]{lec01::def:ordered-field}. For \(x, y \in F\), the \emph{distance} between \(x\) and \(y\) is
  \[
    d(x, y) \defeq \abs{x - y}
  \]
  \par\smallskip\noindent{\footnotesize\texttt{LaTeX: \detokenize{\dist{x}{y}}}}
\end{dfn}

\begin{lem}[Symmetry of Distance]{distance-symmetry}
  Let \(F\) be an \xref[ordered field]{lec01::def:ordered-field}. For all \(x, y \in F\),
  \[
    \dist{x}{y} = \dist{y}{x}
  \]
\end{lem}

\begin{proof}
  By \xref[definition]{def:distance}, \(\dist{x}{y} = \abs{x - y}\) and \(\dist{y}{x} = \abs{y - x} = \abs{-(x - y)}\). It suffices to show \(\abs{-(x - y)} = \abs{x - y}\); we split on the sign of \(x - y\)

  Case 1: \(x - y \geq 0\). Then \(\abs{x - y} = x - y\). Also \(-(x - y) \leq 0\), so \(\abs{-(x - y)} = -(-(x - y)) = x - y = \abs{x - y}\)

  Case 2: \(x - y < 0\). Then \(\abs{x - y} = -(x - y)\). Also \(-(x - y) > 0\), so \(\abs{-(x - y)} = -(x - y) = \abs{x - y}\)
\end{proof}

\begin{lem}[Triangle Inequality for Distance]{triangle-inequality-distance}
  Let \(F\) be an \xref[ordered field]{lec01::def:ordered-field}. For all \(x, y, z \in F\),
  \[
    \dist{x}{z} \leq \dist{x}{y} + \dist{y}{z}
  \]
\end{lem}

\begin{proof}
  Let \(a = x - y\) and \(b = y - z\). Then \(x - z = a + b\), so \(\dist{x}{z} = \abs{a + b}\).
  We split on the sign of \(a + b\)

  Case 1: \(a + b \geq 0\). Then \(\abs{a + b} = a + b\). Since \(a \leq \abs{a}\) and \(b \leq \abs{b}\),
  \[
    \abs{a + b} = a + b \leq \abs{a} + \abs{b} = \dist{x}{y} + \dist{y}{z}
  \]

  Case 2: \(a + b < 0\). Then \(\abs{a + b} = -(a + b) = -a - b\). Since \(-a \leq \abs{a}\) and \(-b \leq \abs{b}\),
  \[
    \abs{a + b} = -a - b \leq \abs{a} + \abs{b} = \dist{x}{y} + \dist{y}{z}
  \]
\end{proof}

\begin{cor}[Metric on a Complete Ordered Field]{complete-ordered-field-metric}
  Let \(F\) be a \xref[complete ordered field]{lec01::def:complete-ordered-field},
  and let \(d\) be the \xref[absolute-value distance]{def:distance} on \(F\).
  Under \xref[the identification with the reals]{lec01::rem:complete-field-identification},
  \((F,d)\) is a \xref[metric space]{def:metric-space}
\end{cor}

\begin{proof}
  Under this identification, the distance is real-valued. By
  \xref[the definition of absolute value]{lec01::def:absolute-value},
  \[
    \dist{x}{y}=\abs{x-y}\geq0,
    \qquad
    \dist{x}{y}=0\iff x-y=0\iff x=y
  \]
  \xref[Symmetry]{lem:distance-symmetry} and
  \xref[the triangle inequality]{lem:triangle-inequality-distance}
  give the remaining two \xref[metric axioms]{def:metric}
\end{proof}

\begin{lem}[\(\varepsilon\)-Criterion for Equality]{epsilon-equality}
  Let \(F\) be an \xref[ordered field]{lec01::def:ordered-field}. For \(a, b \in F\),
  \[
    a=b
    \iff
    \forall \varepsilon \in \apos{F},\, \abs{a-b}<\varepsilon
  \]
\end{lem}

\begin{proof}In order to proof \xref[Epsilon Equality Lemma]{lem:epsilon-equality}, we need to prove both directions

\noindent\textbf{(a)}: $a=b \implies \forall \varepsilon\in\apos{F}, \abs{a-b}<\varepsilon$

Suppose \(a=b\). Then \(\abs{a-b}=0\). For any \(\varepsilon>0\), we have \(\abs{a-b}=0<\varepsilon\)

\noindent\textbf{(b)}: $\forall \varepsilon\in\apos{F}, \abs{a-b}<\varepsilon \implies a=b$

Assume \(a \neq b\)

Then, this means that \[\exists \varepsilon_0\in\apos{F}, \abs{a - b} = \varepsilon_0 \]

Taking \(\varepsilon=\varepsilon_0\) gives \(\varepsilon_0=\abs{a-b}<\varepsilon_0\), a contradiction

So, we have \(\forall \varepsilon\in\apos{F}, \abs{a-b}<\varepsilon \implies \lnot a \neq b\)

Since $\lnot a \neq b \iff a = b$, So, $\forall \varepsilon\in\apos{F}, \abs{a-b}<\varepsilon \implies a=b$

Therefore, both implications hold, and hence proven
\end{proof}

\section{Topological Spaces} 

\begin{dfn}[Topology]
  Let $X$ be a set. A \emph{topology} on $X$, called $\mathcal{T}$, is a subset of $\PowerSet{X}$ satisfying
  \xref[containment of the empty set and the whole set]{axm:topology-empty-and-whole},
  \xref[closure under finite intersections]{axm:topology-finite-intersections}, and
  \xref[closure under arbitrary unions]{axm:topology-arbitrary-unions}
\end{dfn}

\begin{axm}{topology-empty-and-whole}
If $\mathcal T$ is a topology on $X$, then
\[\emptyset, X \in \mathcal T\]
\end{axm}

\begin{axm}{topology-finite-intersections}
If $U, V \in \mathcal T$, then 
\[U \cap V \in \mathcal T\] 
\end{axm}

\begin{axm}{topology-arbitrary-unions}
If $U: I \to \mathcal T$ is a family of sets in $\mathcal T$, then 
\[\bigcup_{I} U \in \mathcal T\] 
\end{axm}

\begin{dfn}[Topological Spaces]{topological-space}
  A \emph{topological space} is a pair \((X,\mathcal T)\), where \(X\) is a set and
  \(\mathcal T\) is a \xref[topology]{def:topology} on \(X\)
\end{dfn} 

\begin{dfn}[Induced Topology]{induced-topology}
  Let $(X, d)$ be a metric space, where we define a \emph{induced topology} $\mathcal T_d$ induced by $d$ ;s 
  \[
    \mathcal T_d\defeq
    \left\{U\in\PowerSet(X)\;\middle|\;
      \forall x\in U,\,\exists r\in\apos{\RR},\,
      \ball{r}{x}\subseteq U\right\}
  \]

\end{dfn} 

\begin{thm}[Induced Topology is a Topology]
  \[
    (X,d)\text{ is a metric space}
    \implies
    (X,\mathcal T_d)\text{ is a topological space}
  \]
\end{thm} 

\begin{proof}
  By definition, \(\mathcal T_d\subseteq\PowerSet(X)\).
  The lemmas on \xref[the empty set and the whole space]{lem:metric-empty-and-whole-open},
  \xref[finite intersections]{lem:metric-finite-intersections}, and
  \xref[arbitrary unions]{lem:metric-arbitrary-unions} give
  \[
    \emptyset,X\in\mathcal T_d,
    \qquad
    U,V\in\mathcal T_d\implies U\cap V\in\mathcal T_d,
    \qquad
    U:I\to\mathcal T_d\implies\bigcup_I U\in\mathcal T_d
  \]
  Thus \(\mathcal T_d\) satisfies the three
  \xref[topology axioms]{def:topology}, so \((X,\mathcal T_d)\)
  is a \xref[topological space]{def:topological-space}
\end{proof}

\section{Open and Closed Sets}

\begin{dfn}[Open Sets]
  A set $U \subseteq X$ is \emph{open} in a topological space $(X, \mathcal{T})$ if $U \in \mathcal{T}$. 
\end{dfn}

\begin{dfn}[Closed Sets]
  A set $C \subseteq X$ is \emph{closed} in a topological space $(X, \mathcal{T})$ if its complement $X \setminus C$ is open, i.e., $X \setminus C \in \mathcal{T}$. 
\end{dfn}

\begin{cor}[Arbitrary Unions of Open Sets]{open-arbitrary-unions}
  Let \((X,\mathcal T)\) be a
  \xref[topological space]{def:topological-space}, and let
  \(U:I\to\PowerSet(X)\). If every \(U(i)\) is
  \xref[open]{def:open-sets}, then
  \[
    \bigcup(U)\in\mathcal T
  \]
\end{cor}

\begin{proof}
  Each \(U(i)\in\mathcal T\) by the definition of an open set, so
  \xref[the arbitrary-union axiom]{axm:topology-arbitrary-unions}
  gives \(\bigcup(U)\in\mathcal T\)
\end{proof}

\begin{cor}[Finite Intersections of Open Sets]{open-finite-intersections}
  Let \((X,\mathcal T)\) be a
  \xref[topological space]{def:topological-space}, and let
  \(U:I\to\PowerSet(X)\), where \(I\) is finite and nonempty.
  If every \(U(i)\) is \xref[open]{def:open-sets}, then
  \[
    \bigcap(U)\in\mathcal T
  \]
\end{cor}

\begin{proof}
  A single member belongs to \(\mathcal T\). Applying
  \xref[the intersection axiom]{axm:topology-finite-intersections}
  repeatedly proves the claim by induction on the number of sets
\end{proof}

\begin{cor}[Arbitrary Intersections of Closed Sets]{closed-arbitrary-intersections}
  In a \xref[topological space]{def:topological-space}
  \((X,\mathcal T)\), let
  \(C:I\to\PowerSet(X)\), where \(I\) is nonempty.
  If every \(C(i)\) is \xref[closed]{def:closed-sets}, then
  \(\bigcap(C)\) is closed
\end{cor}

\begin{proof}
  Each \(X\setminus C(i)\) is open by the definition of a closed set.
  By \xref[De Morgan's laws]{lec01::thm:de-morgan-laws} and
  \xref[the open-union corollary]{cor:open-arbitrary-unions},
  \[
    X\setminus\bigcap(C)
    =\bigcup(i\map X\setminus C(i))\in\mathcal T
  \]
  Thus \(\bigcap(C)\) is closed
\end{proof}

\begin{cor}[Finite Unions of Closed Sets]{closed-finite-unions}
  Let \((X,\mathcal T)\) be a
  \xref[topological space]{def:topological-space}, and let
  \(C:I\to\PowerSet(X)\), where \(I\) is finite.
  If every \(C(i)\) is \xref[closed]{def:closed-sets}, then
  \(\bigcup(C)\) is closed
\end{cor}

\begin{proof}
  If \(I\) is empty, then \(\bigcup(C)=\emptyset\) is closed since
  \(X\in\mathcal T\). Otherwise, each \(X\setminus C(i)\) is open.
  By \xref[De Morgan's laws]{lec01::thm:de-morgan-laws} and
  \xref[the open-intersection corollary]{cor:open-finite-intersections},
  \[
    X\setminus\bigcup(C)
    =\bigcap(i\map X\setminus C(i))\in\mathcal T
  \]
  Thus \(\bigcup(C)\) is closed
\end{proof}

\begin{prop}[Open Balls are Open]{metric-open-balls-open}
  In a \xref[metric space]{def:metric-space} \((X,d)\), every
  \xref[open ball]{def:open-balls} is \xref[open]{def:open-sets} in the
  \xref[induced topology]{def:induced-topology}
\end{prop}

\begin{proof}
  Let \(z\in\ball{r}{x}\), and choose \(s=r-d(x,z)>0\).
  For \(y\in\ball{s}{z}\), the
  \xref[triangle inequality]{axm:metric-triangle-inequality} gives
  \[
    d(x,y)\leq d(x,z)+d(z,y)<d(x,z)+s=r
  \]
  Thus \(\ball{s}{z}\subseteq\ball{r}{x}\), so \(\ball{r}{x}\) is open
\end{proof}

\section{Limit Points and Isolated Points}

\begin{dfn}[Limit Points]{limit-points}
  Let \((X,d)\) be a \xref[metric space]{def:metric-space}.
  A \emph{limit point} \(x\in X\) of a set \(U\subseteq X\) is a point such that
  \[
    \forall\varepsilon\in\apos{\RR},\quad
    \ball{\varepsilon}{x}\cap\bigl(U\setminus\{x\}\bigr)\neq\emptyset
  \]
\end{dfn}

\begin{rem}[Limit Point Intuition]{limit-point-intuition}
  A \xref[limit point]{def:limit-points} may lie inside or outside the set:
  no matter how small a ball centred at it is, that ball contains a point
  of the set different from the centre itself
\end{rem}

\begin{lem}[Balls Disjoint from a Set]{ball-disjoint-limit-points}
  Let \((X,d)\) be a \xref[metric space]{def:metric-space}, let
  \(A\subseteq X\), and let \(L\) be the set of
  \xref[limit points]{def:limit-points} of \(A\).
  For \(x\in X\) and \(r\in\apos{\RR}\),
  \[
    \ball{r}{x}\cap A=\emptyset
    \implies\ball{r}{x}\cap L=\emptyset
  \]
\end{lem}

\begin{proof}
  If point $\ell$ is a limit point of $A$, then 
  \[\forall \varepsilon \in \apos{\RR} , \, \ball{\varepsilon}{\ell} \cap (A \setminus \{\ell\}) \neq \emptyset \] 

  Since 
  \(\ball{r}{x} \cap A = \emptyset\), 
  Assume such $\ell$ exist in \(\ball{r}{x}\), then $\ell \in \ball{r}{x}$, which means  
  \[d(\ell,x) < r\] 

  Then, choose $\varepsilon = r - d(\ell,x) > 0$. For any \(y\in\ball{\varepsilon}{\ell}\), the
  \xref[triangle inequality]{axm:metric-triangle-inequality} and
  \xref[symmetry]{axm:metric-symmetry} give
  \[
    d(x,y)\leq d(x,\ell)+d(\ell,y)
    <d(\ell,x)+\varepsilon=r
  \]
  Thus we have
  \(\ball{\varepsilon}{\ell} \subseteq \ball{r}{x}\) 

  However we know that \(\ball{r}{x} \cap A = \emptyset\), which means
  \(\ball{\varepsilon}{\ell} \cap A = \emptyset \) 

  Which creates an contradiction. So, there does not exist a limit point $\ell$ in \(\ball{r}{x}\), which means
  \(\ball{r}{x} \cap L = \emptyset\) 
\end{proof}

  Let \((X,d)\) be a \xref[metric space]{def:metric-space}.

  \[
    \exists\varepsilon\in\apos{\RR},\quad

\begin{dfn}[Closure]{closure}
  Let \((X,\mathcal T)\) be a
  \xref[topological space]{def:topological-space}.
  The \emph{closure function}
  \(\closure:\PowerSet(X)\to\PowerSet(X)\) assigns to each
  \(A\subseteq X\) the smallest \xref[closed]{def:closed-sets} subset

      A\subseteq C,\quad C\text{ is closed}\\
      \forall D\subseteq X,\quad
      (A\subseteq D\text{ and }D\text{ is closed})\implies C\subseteq D
    \end{cases}
  \]

\end{dfn}

\begin{lem}[Closed Supersets Contain Limit Points]{closure-from-limit-points}
  Let \((X,d)\) be a \xref[metric space]{def:metric-space}, let
  \(A\subseteq X\), and let \(L\) be the set of
  \xref[limit points]{def:limit-points} of \(A\).
  Let \(D\subseteq X\) be \xref[closed]{def:closed-sets} in the
  \xref[induced topology]{def:induced-topology}. Then
  \[
    A\subseteq D\implies L\subseteq D
  \]
\end{lem}

\begin{proof}
  Let $D$ be a \xref[closed set]{def:closed-sets} containing $A$. Which means $E \defeq X \setminus D$ is \xref[open]{def:open-sets}. 

  Since $\ell \in L$ is a \xref[limit point]{def:limit-points} of $A$, we know that 
  \[\forall \varepsilon \in \apos{\RR} , \, \ball{\varepsilon}{\ell} \cap A \setminus \{\ell\} \neq \emptyset\]. 
  Suppose $\exists \ell \in L, \, \ell \notin D, \, \ell \notin A$. This means $\ell \in E$. 
  
  Since all element which is in $A$ not in $E$, and since for such $\ell$ we already said
  \[\ball{\varepsilon}{\ell} \cap A \setminus \{\ell\} \neq \emptyset\]
  Then, 
  \[\forall \varepsilon \in \apos{\RR} , \, \exists \delta \in \ball{\varepsilon}{\ell} , \, \delta \in A \implies \ball{\varepsilon}{\ell} \text{ is not a subset of } E\] 

  Which means that there does not exist a $r$ such that $\ball{r}{\ell} \subseteq E$, contradicts with the requirement where $E$ must be an \xref[open set]{def:induced-topology} 
\end{proof}

\begin{lem}[Adjoining Limit Points Gives a Closed Set]{adjoining-limit-points}
  Let \((X,d)\) be a \xref[metric space]{def:metric-space}, let
  \(A\subseteq X\), and let \(L\) be the set of
  \xref[limit points]{def:limit-points} of \(A\).
  Then \(A\cup L\) is \xref[closed]{def:closed-sets} in the
  \xref[induced topology]{def:induced-topology}
\end{lem}

\begin{proof}
  Define $C \defeq X \setminus (A \cup L)$. Choose $x \in C$. It must not be a limit point of $A$. Then, 
  
  By definition of limit point, we know that 
  \[\forall x \in C, \, \exists r \in \apos{\RR}, \, \ball{r}{x} \setminus \{x\} \cap A = \emptyset\] 
  Since $x \notin A$, simplify, 
  \[\forall x \in C, \, \exists r \in \apos{\RR}, \, \ball{r}{x} \cap A = \emptyset\] 

  By \xref[the ball lemma]{lem:ball-disjoint-limit-points},
  \(\ball{r}{x}\cap L=\emptyset\), so \(\ball{r}{x}\subseteq C\).
  Since \(x\in C\) was arbitrary, \(C\) is open by the
  \xref[definition of the induced topology]{def:induced-topology}.
  Hence \(A\cup L\) is closed
\end{proof}

\begin{thm}[Closure by Adjoining Limit Points]{closure-limit-points}
  Let \((X,d)\) be a \xref[metric space]{def:metric-space}, and let
  \(A\subseteq X\). With closure taken in the
  \xref[induced topology]{def:induced-topology}, write
  \[
    L\defeq\{x\in X:x\text{ is a \xref[limit point]{def:limit-points} of }A\}
  \]
  Then
  \[
    \cl{A}=A\cup L
  \]
\end{thm}

\begin{proof}
  By \xref[the lemma on adjoining limit points]{lem:adjoining-limit-points},
  \(A\cup L\) is closed and contains \(A\), so the
  \xref[definition of closure]{def:closure} gives
  \(\cl{A}\subseteq A\cup L\).
  Conversely, \(\cl{A}\) is a closed set containing \(A\), so
  \xref[the closed-superset lemma]{lem:closure-from-limit-points}
  gives \(L\subseteq\cl{A}\). Hence \(A\cup L\subseteq\cl{A}\),
  proving equality
\end{proof}

\begin{rem}[Adding Limit Points Once is Enough]{closure-limit-points-intuition}
  Start with \(A\) and add all its limit points to obtain \(A\cup L\).
  This set is already the \xref[closure]{def:closure} of \(A\), so it is
  closed and contains all its own limit points. No further points need
  to be added, so repeating the process leaves \(A\cup L\) unchanged
\end{rem}

\begin{dfn}[Cover]{cover}
  Let \(X\) be a set and \(S\subseteq X\).
  A \emph{cover} of \(S\) is a family \(U:I\to\PowerSet(X)\) such that
  \[
    S\subseteq\bigcup_I U
  \]
\end{dfn}

\begin{dfn}[Finite Cover]{finite-cover}
  A \xref[cover]{def:cover} \(U:I\to\PowerSet(X)\) of \(S\subseteq X\)
  is \emph{finite} if its index set \(I\) is finite, equivalently,
  \[
    \card{I}<\card{\NN}
  \]
\end{dfn}

\begin{dfn}[Open Cover]{open-cover}
  Let \((X,\mathcal T)\) be a
  \xref[topological space]{def:topological-space}, and let \(S\subseteq X\).
  A \xref[cover]{def:cover} \(U:I\to\PowerSet(X)\) of \(S\) is an
  \emph{open cover} if
  \[
    \forall i\in I,\quad U(i)\in\mathcal T
  \]
\end{dfn}

\begin{dfn}[Subcover]{subcover}
  Let \(U:I\to\PowerSet(X)\) be a \xref[cover]{def:cover} of
  \(S\subseteq X\). A \emph{subcover} of \(U\) is a cover
  \(V:J\to\PowerSet(X)\) of \(S\) such that
  \[
    \forall j\in J,\,\exists i\in I,\quad V(j)=U(i)
  \]
  The subcover is \emph{finite} if \(V\) is a
  \xref[finite cover]{def:finite-cover}
\end{dfn}

\begin{dfn}[Compact Set]{compact-set}
  Let \((X,\mathcal T)\) be a
  \xref[topological space]{def:topological-space}.
  A subset \(S\subseteq X\) is \emph{compact} if every
  \xref[open cover]{def:open-cover} of \(S\) has a finite
  \xref[subcover]{def:subcover}
\end{dfn}

\section{The Usual Metric and Topology on \texorpdfstring{\(\RR\)}{R}}

\begin{dfn}[Usual Topology on the Real Line]{usual-topology-real}
  Equip \(\RR\) with the
  \xref[absolute-value distance]{def:distance}, which is a
  \xref[metric]{cor:complete-ordered-field-metric}:
  \[
    \dist{x}{y}=\abs{x-y},\qquad x,y\in\RR
  \]

  The \emph{usual topology} on \(\RR\) is the
  \xref[topology induced]{def:induced-topology} by the absolute-value metric
  \(d\). In this section, balls and open sets in \(\RR\) refer to this
  metric and topology
\end{dfn}

\begin{prop}[Open Balls on the Real Line]{real-open-balls}
  For \(x\in\RR\) and \(r\in\apos{\RR}\), the
  \xref[open ball]{def:open-balls} is the
  \xref[open interval]{lec02::def:open-interval}
  \[
    \ball{r}{x}=(x-r,x+r)
  \]
\end{prop}

\begin{proof}
  For \(y\in\RR\),
  \[
    y\in\ball{r}{x}
    \iff\abs{x-y}<r
    \iff x-r<y<x+r
    \iff y\in(x-r,x+r)
  \]
\end{proof}

\begin{cor}[Open Intervals are Open]{real-open-intervals}
  For \(a,b\in\RR\) with \(a<b\), the
  \xref[open interval]{lec02::def:open-interval} \((a,b)\) is
  \xref[open]{def:open-sets} in the
  \xref[usual topology]{def:usual-topology-real}
\end{cor}

\begin{proof}
  For each \(x\in(a,b)\), we can choose a ball centred at \(x\)
  with radius \(r=\min(x-a,b-x)>0\). Then
  \[
    \ball{r}{x}=(x-r,x+r)\subseteq(a,b)
  \]
  Thus every point of \((a,b)\) has a ball contained in \((a,b)\),
  so \((a,b)\) is open by the
  \xref[definition of the induced topology]{def:induced-topology}
\end{proof}

\begin{prop}[Open Sets as Unions of Open Intervals]{real-open-sets-intervals}
  A set \(U\subseteq\RR\) is \xref[open]{def:open-sets} in the
  \xref[usual topology]{def:usual-topology-real} if and only if it is a
  union of \xref[open intervals]{lec02::def:open-interval}
\end{prop}

\begin{proof}
  \noindent\textbf{(\(\Rightarrow\))}
  Suppose \(U\) is open. Let
  \[
    I\defeq\{(x,r)\in U\times\apos{\RR}:\ball{r}{x}\subseteq U\},
    \qquad
    J:I\to\PowerSet(\RR)\defeq(x,r)\map(x-r,x+r)
  \]
  Every \(J(x,r)=\ball{r}{x}\) is contained in \(U\), and each
  \(x\in U\) belongs to one such interval by the definition of
  \(\mathcal T_d\). Hence \(U=\bigcup(J)\)

  \noindent\textbf{(\(\Leftarrow\))}
  Each \xref[open interval is open]{cor:real-open-intervals}, and
  \xref[arbitrary unions of open sets are open]{cor:open-arbitrary-unions}
\end{proof}

\begin{prop}[Closed Intervals are Closed]{real-closed-intervals}
  For \(a,b\in\RR\) with \(a\leq b\), the
  \xref[closed interval]{lec02::def:closed-interval} \([a,b]\) is
  \xref[closed]{def:closed-sets} in the
  \xref[usual topology]{def:usual-topology-real}
\end{prop}

\begin{proof}
  Let \(x\in\RR\setminus[a,b]\). If \(x<a\), choose \(r=a-x>0\);
  if \(x>b\), choose \(r=x-b>0\). In either case, the ball centred at
  \(x\) satisfies
  \[
    \ball{r}{x}=(x-r,x+r)\subseteq\RR\setminus[a,b]
  \]
  Thus \(\RR\setminus[a,b]\) is open, so \([a,b]\) is closed by the
  \xref[definition of a closed set]{def:closed-sets}
\end{proof}

\begin{dfn}[Cantor Stages]{cantor-stages}
  The function \(A:\NN\to\PowerSet(\RR)\) is defined recursively by
  \[
    A(0)\defeq[0,1],
    \qquad
    A(n+1)\defeq
    \left\{\frac{x}{3}:x\in A(n)\right\}
    \cup
    \left\{\frac{x+2}{3}:x\in A(n)\right\}
  \]
  Each stage removes the open middle third of every remaining interval,
  keeping its endpoints
\end{dfn}

\begin{eg}[The First Cantor Stages]{cantor-first-stages}
  \[
    \begin{aligned}
      A(1)&=[0,\tfrac13]\cup[\tfrac23,1]\\
      A(2)&=[0,\tfrac19]\cup[\tfrac29,\tfrac13]
             \cup[\tfrac23,\tfrac79]\cup[\tfrac89,1]
    \end{aligned}
  \]
\end{eg}

\begin{dfn}[Middle-Third Cantor Set]{cantor-set}
