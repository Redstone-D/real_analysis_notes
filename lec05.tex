\section{Subsequence}

\begin{dfn}[Subsequence]{subsequence}
If a sequence $n: \NNOne \to \NNOne$ is \xref[strictly increasing]{lec04::def:strictly-increasing-sequence}, then the sequence $b:\NNOne\to F\defeq k\map a(n(k))$ is called a \emph{subsequence} of $a: \NNOne \to F$
\end{dfn}

\begin{thm}[Subsequence Index Grows]{subsequence-index-grows}
If $b:\NNOne\to F\defeq k\map a(n(k))$ is a subsequence of $a: \NNOne \to F$, then $\forall k \in \NNOne, n(k) \geq k$
\end{thm}

\begin{proof}
Since $n: \NNOne \to \NNOne$ is strictly increasing, we have $n(1) < n(2) < n(3) < \cdots$

Notice that $n(1) \geq 1$, since $n(1) \in \NNOne$

Also, if $n(k) \geq k$, then $n(k + 1) > n(k) \geq k$, so $n(k + 1) \geq k + 1$

By induction, we have $\forall k \in \NNOne, n(k) \geq k$
\end{proof}

\begin{thm}[Subsequence of a Subsequence]{subsequence-of-subsequence}
A subsequence of a subsequence is still a subsequence of the original sequence
\end{thm}

\begin{rem}[Intuition: Picking Forward]{subsequence-intuition}
A subsequence can be understand as picking element from a sequence, however during picking we can only go forward, cannot go back to pick an element or pick an element twice
\end{rem}

\section{Series}

\begin{dfn}[Series Operator]{series-operator}
  Let $F$ be a \xref[field]{lec01::def:field}. The \emph{series operator}
  \[
    \mathrm{series} : (\NNOne \to F) \to (\NNOne \to F)
  \]
  is defined recursively by
  \[
    \mathrm{series}(a)(1) \defeq a(1),
    \qquad
    \mathrm{series}(a)(n + 1) \defeq \mathrm{series}(a)(n) + a(n + 1)
  \]
\end{dfn}

\begin{dfn}[Series]{series}
  Let $F$ be a \xref[field]{lec01::def:field} and $a : \NNOne \to F$. The \emph{series of $a$} is the sequence
  \[
    \mathrm{series}(a) : \NNOne \to F,
  \]
  obtained by applying the \xref[series operator]{def:series-operator} to $a$. Its $n$-th term is the partial sum $\sum_{1}^{n} a$
\end{dfn} 

\begin{dfn}[Convergent Series]{convergent-series}
  Let $F$ be a \xref[field]{lec01::def:field} and $a : \NNOne \to F$. The \xref[series of $a$]{def:series} is \emph{convergent} if $\mathrm{series}(a)$ is a \xref[convergent sequence]{lec04::def:convergent-sequence}
\end{dfn}

\begin{dfn}[Infinite Summation]{infinite-series}
  Let $F$ be a \xref[field]{lec01::def:field} and $a : \NNOne \to F$ with a \xref[convergent series]{def:convergent-series}. The \emph{infinite summation} of $a$ is
  \[
    \infsum a \defeq \seqlim(\mathrm{series}(a))
  \]
  \par\smallskip\noindent{\footnotesize\texttt{LaTeX: \detokenize{\infsum(a)}}}
\end{dfn} 

\begin{thm}[Linearity of Infinite Summation]{linearity-of-infinite-summation}
  Let $F$ be a \xref[field]{lec01::def:field} and $a, b : \NNOne \to F$ with convergent series. Then
  \[
    \infsum (a + b) = \infsum a + \infsum b,
    \qquad
    \infsum (\lambda a) = \lambda \infsum a
  \]
\end{thm}

\section{Comparison Test for Sequence}

\begin{lem}[Series with Non-Negative Terms is Increasing]{series-nonneg-increasing}
  Let $F$ be an \xref[ordered field]{lec01::def:ordered-field} and $a : \NNOne \to F$ with $a(n) \geq 0$ for all $n \in \NNOne$. Then $\mathrm{series}(a)$ is an \xref[increasing sequence]{lec04::def:increasing-sequence}
\end{lem}

\begin{lem}[Series with Positive Terms is Strictly Increasing]{series-positive-strictly-increasing}
  Let $F$ be an \xref[ordered field]{lec01::def:ordered-field} and $a : \NNOne \to F$ with $a(n) > 0$ for all $n \in \NNOne$. Then $\mathrm{series}(a)$ is a \xref[strictly increasing sequence]{lec04::def:strictly-increasing-sequence}
\end{lem}

\begin{cor}[Comparison Test for Series]{comparison-test}
  Let $F$ be a \xref[complete ordered field]{lec01::def:complete-ordered-field}, $a, b : \NNOne \to F$ with $0 \leq a_n \leq b_n$ for all $n \in \NNOne$. If the \xref[series of $b$]{def:series} is \xref[convergent]{def:convergent-series}, then the series of $a$ is convergent and $\infsum a \leq \infsum b$. Follows by applying \xref{lec04::thm:monotone-comparison-increasing} to the partial sum sequences $\mathrm{series}(a)$ and $\mathrm{series}(b)$, both of which are increasing by \xref{lem:series-nonneg-increasing}
\end{cor}

\begin{thm}[Absolute Convergence Implies Convergence]{absolute-convergence}
  Let $F$ be a \xref[complete ordered field]{lec01::def:complete-ordered-field} and $a : \NNOne \to F$. If the series of $|a|$ (where $|a|(n) \defeq \abs{a(n)}$) is \xref[convergent]{def:convergent-series}, then the \xref[series of $a$]{def:series} is convergent
\end{thm}

\begin{proof}
  Since \(0\leq a(n)+\abs{a(n)}\leq2\abs{a(n)}\), the
  \xref[comparison test]{cor:comparison-test} gives convergence of the
  series of \(n\map a(n)+\abs{a(n)}\), and subtracting the convergent
  series of \(n\map\abs{a(n)}\) gives convergence of the series of \(a\)
  by \xref[linearity]{thm:linearity-of-infinite-summation}
\end{proof}

\begin{thm}[Convergence of Nonnegative Series]{nonnegative-series-criterion}
  Let \(F\) be a
  \xref[complete ordered field]{lec01::def:complete-ordered-field}, and let
  \(a:\NNOne\to F\) satisfy \(a(n)\geq0\) for every \(n\in\NNOne\).
  
  The \xref[series of \(a\)]{def:series} is
  \xref[convergent]{def:convergent-series} if and only if \(P\) is
  \xref[bounded above]{lec01::def:bounded-above} in \(F\).
  In that case,
  \[
    \infsum(a)=\sup(P), \quad P\defeq\left\{\sum_1^n(a):n\in\NNOne\right\}
  \]
\end{thm}

\begin{proof}

  Suppose \(P\) is bounded above and \(\mathrm{series}(a)\) converges.
  Write \(M=\sup(P)\) and \(L=\seqlim(\mathrm{series}(a))\).
  According to the
  \xref[\(\varepsilon\)-criterion for the supremum]{lec01::prop:epsilon-supremum},
  \[
    \forall\varepsilon_1\in\apos{F},\,\exists s\in P,\quad
    \abs{M-s}<\varepsilon_1
  \]
  According to the \xref[definition of a limit]{lec04::def:limit},
  \[
    \forall\varepsilon_2\in\apos{F},\,\exists N\in\NNOne,\,
    \forall n\in\NNOne,\quad
    n\geq N\implies\abs{\mathrm{series}(a)(n)-L}<\varepsilon_2
  \]
  Fix arbitrary \(\varepsilon_1,\varepsilon_2\in\apos{F}\) and
  corresponding witnesses \(s,N\), and let
  \(\varepsilon=\max(\varepsilon_1,\varepsilon_2)\).
  \[\forall s \in P, \, \exists k:\NNOne, \, s=\mathrm{series}(a)(k) \]
  Set \(n=\max(k,N)\). Since the partial sums are increasing and
  \(M\) is an upper bound,
  \[
    s=\mathrm{series}(a)(k)\leq\mathrm{series}(a)(n)\leq M
  \]
  Hence
  \[
    \abs{M-\mathrm{series}(a)(n)}
    =M-\mathrm{series}(a)(n)
    \leq M-s<\varepsilon_1\leq\varepsilon
  \]
  Also \(n\geq N\), so the convergence bound applies at this same index

  Thus
  \[
    \abs{M-\mathrm{series}(a)(n)}<\varepsilon,
    \qquad
    \abs{\mathrm{series}(a)(n)-L}<\varepsilon
  \]
  By the \xref[triangle inequality]{lec03::lem:triangle-inequality-distance},
  \[
    \abs{M-L}
    \leq\abs{M-\mathrm{series}(a)(n)}
      +\abs{\mathrm{series}(a)(n)-L}
    <2\cdot\varepsilon
  \]
  The function \(g(\varepsilon)\defeq2\cdot\varepsilon\) is a
  \xref[linear challenge function]{lec04::lem:linear-is-challenge}.
  Taking \(\varepsilon_1=\varepsilon_2=\varepsilon\) shows that
  \(\abs{M-L}<g(\varepsilon)\) for every \(\varepsilon\in\apos{F}\).
  By the \xref[equality criterion with a challenge function]{lec04::cor:equality-challenge-replacement},
  \[
    \sup(P)=\seqlim(\mathrm{series}(a))=\infsum(a)
  \]
\end{proof}

\begin{rem}[Two Methods for Checking Series Convergence]{two-methods-series}
  Depending on the sign structure of the terms of $a$:
  \begin{itemize}[nosep]
    \item \emph{Single-sign} (all $a(n) \geq 0$ or all $a(n) \leq 0$): apply the \xref[Comparison Test for Series]{cor:comparison-test} directly. Monotonicity of $\mathrm{series}(a)$ comes from \xref{lem:series-nonneg-increasing} (or the sign-flipped version)
    \item \emph{Mixed-sign}: apply \xref[Absolute Convergence]{thm:absolute-convergence} first, reducing the question to $\mathrm{series}(|a|)$ (non-negative), where the Comparison Test applies. For alternating terms, the \xref[alternating-series test]{thm:alternating-series-test} may apply even when the series of absolute values diverges
  \end{itemize}
\end{rem} 

\section{Grouping of Series}

\begin{dfn}[Grouping Index]{grouping-index}
  A \emph{grouping index} is a sequence \(i:\NNOne\to\NNOne\) satisfying
  \[
    i(1)=1,\qquad\forall n\in\NNOne,\quad i(n)<i(n+1)
  \]
\end{dfn}

\begin{dfn}[Grouping of Series]{series-grouping}
  Let \(F\) be a \xref[field]{lec01::def:field} and \(i\) a
  \xref[grouping index]{def:grouping-index}. Define
  \[
    \mathrm{group}(i):(\NNOne\to F)\to(\NNOne\to F)
    \defeq a\map n\map\sum_{i(n)}^{i(n+1)-1}(a)
  \]
  The grouped series of \(a\) is the \xref[series]{def:series} of
  \(\mathrm{group}(i)(a)\)
\end{dfn}

\begin{lem}[Partial Sums of a Grouped Series]{grouped-partial-sums}
  Let \(F\) be a \xref[field]{lec01::def:field}, \(i\) a
  \xref[grouping index]{def:grouping-index}, and \(a:\NNOne\to F\).
  Then
  \[
    \forall m\in\NNOne,\quad
    \sum_1^m\bigl(\mathrm{group}(i)(a)\bigr)
    =\sum_1^{i(m+1)-1}(a)
  \]
\end{lem}

\begin{proof}
  By the \xref[definition of grouping]{def:series-grouping} and repeated
  application of \xref[splitting of summation]{lec01::prop:summation-split},
  the first \(m\) blocks combine into the sum from \(i(1)=1\) to
  \(i(m+1)-1\)
\end{proof}
