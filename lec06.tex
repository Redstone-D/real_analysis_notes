\section{Limit of Functions} 

\begin{dfn}[Existence of a Limit at a Point]{limit-of-functions}
  Let \((X,d_X)\) and \((B,d_B)\) be
  \xref[metric spaces]{lec03::def:metric-space}, let \(A\subseteq X\)
  and \(f:A\to B\), and let \(c\in X\) be a
  \xref[limit point]{lec03::def:limit-points} of \(A\)

  If there exists a point \(\ell \in B\) such that \[\forall \varepsilon \in \apos{\RR}, \, \exists \delta \in \apos{\RR}, \, \forall x \in A, \, 0 < d_X(x,c) < \delta \implies d_B(f(x),\ell) < \varepsilon\] then we say that \(f\) has a limit at \(c\)
\end{dfn}

\begin{thm}[Uniqueness of a Function Limit]{function-limit-uniqueness}
  With the setup of the \xref[existence definition]{def:limit-of-functions},
  if a function $f$ has a limit at a point \(c\), then the limit is unique, ie
  \[\exists \ell : B, \, \forall \varepsilon \in \apos{\RR}, \, \exists \delta \in \apos{\RR}, \, \forall x \in A, \, 0 < d_X(x,c) < \delta \implies d_B(f(x),\ell) < \varepsilon\]
  \[\iff\]
  \[\exists! \ell : B, \, \forall \varepsilon \in \apos{\RR}, \, \exists \delta \in \apos{\RR}, \, \forall x \in A, \, 0 < d_X(x,c) < \delta \implies d_B(f(x),\ell) < \varepsilon\]
\end{thm}

\begin{proof}
  Assume that \[\exists \ell_1, \, \forall \varepsilon \in \apos{\RR}, \, \exists \delta_1 \in \apos{\RR}, \, \forall x \in A, \, 0 < d_X(x,c) < \delta_1 \implies d_B(f(x),\ell_1) < \varepsilon\]
   \[\exists \ell_2, \, \forall \varepsilon \in \apos{\RR}, \, \exists \delta_2 \in \apos{\RR}, \, \forall x \in A, \, 0 < d_X(x,c) < \delta_2 \implies d_B(f(x),\ell_2) < \varepsilon\] where $\ell_1 \neq \ell_2$

  Fix \(\varepsilon>0\) and choose the corresponding \(\delta_1,\delta_2\).
  Since \(c\) is a \xref[limit point]{lec03::def:limit-points} of \(A\), choose
  \(x\in A\cap\ball{\min(\delta_1,\delta_2)}{c}\setminus\{c\}\)

  We know that 
  \[\forall x \in A\cap\ball{\min(\delta_1, \delta_2)}{c}\setminus\{c\}, \, d_B(f(x),\ell_1) + d_B(f(x),\ell_2) < 2\varepsilon\]

  Then, according to \xref[triangle inequality]{lec03::axm:metric-triangle-inequality}, 
  \[d_B(\ell_1, \ell_2) \leq d_B(f(x),\ell_1) + d_B(f(x),\ell_2) < 2 \varepsilon\]

  According to \xref[epsilon criteria]{lec04::cor:equality-challenge-replacement}, \[d_B(\ell_1, \ell_2) = 0\] ie \xref[$\ell_1 = \ell_2$]{lec03::axm:metric-separation}. Contradiction. So uniqueness is proven.
\end{proof}

\begin{rem}[Board Comparison]{sequence-function-limit-challenges}
  For a sequence \(a:\NNOne\to X\) in a
  \xref[metric space]{lec03::def:metric-space} \((X,d)\) and a function
  \(f:A\to B\), compare the two challenges
  \begin{center}
    \renewcommand{\arraystretch}{1.25}
    \begin{tabular}{@{}lcc@{}}
       & \textbf{Sequence} & \textbf{Function} \\
      \hline
      Arbitrary tolerance
        & \(\forall\varepsilon\in\apos{\RR}\)
        & \(\forall\varepsilon\in\apos{\RR}\) \\
      Restrictor
        & \(\exists N\in\NNOne\)
        & \(\exists\delta\in\apos{\RR}\) \\
      Input restriction
        & \(n\geq N\)
        & \(0<d_X(x,c)<\delta\) \\
      Output requirement
        & \(d(a(n),\ell)<\varepsilon\)
        & \(d_B(f(x),\ell)<\varepsilon\) \\
    \end{tabular}
  \end{center}
  The output requirement must hold for every \(n\in\NNOne\) or
  \(x\in A\) satisfying its input restriction

  \begin{enumerate}[nosep]
    \item \(\varepsilon\) is arbitrary small
    \item \(\delta\) and \(N\) is used to restrict \(x\)
      st \(x\) is close enough to the limit
    \item \(\delta\) and \(N\) are all depends on \(\varepsilon\)
  \end{enumerate}
\end{rem}

\begin{conv}[The \(\varepsilon\)--\(\delta\) Challenge]{epsilon-delta-challenge}
  A value \(\ell\in B\) passes the
  \emph{\(\varepsilon\)--\(\delta\) challenge} for \(f\) at \(c\) if,
  for any proposed \(\varepsilon>0\), there exists a radius \(\delta>0\)
  such that for all \(x\in A\) with \(0<d_X(x,c)<\delta\),
  \(d_B(f(x),\ell)<\varepsilon\)
\end{conv}

\begin{thm}[Function Limit Challenge Replacement]{function-limit-challenge-replacement}
  Let \((X,d_X)\) and \((B,d_B)\) be
  \xref[metric spaces]{lec03::def:metric-space}, let \(A\subseteq X\)
  and \(f:A\to B\),
  let \(c\in X\) be a \xref[limit point]{lec03::def:limit-points} of \(A\),
  and fix \(\ell\in B\). Let \(g\) be a
  \xref[convergence challenge function]{lec04::def:convergence-challenge-function}.
  With balls taken in \((X,d_X)\),
  \[
    \begin{gathered}
      \bigl(\forall\varepsilon\in\apos{\RR},\,
      \exists\delta\in\apos{\RR},\,
      \forall x\in A\cap\ball{\delta}{c}\setminus\{c\},\quad
      d_B(f(x),\ell)<\varepsilon\bigr)\\
      \iff\\
      \bigl(\forall\varepsilon\in\apos{\RR},\,
      \exists\delta\in\apos{\RR},\,
      \forall x\in A\cap\ball{\delta}{c}\setminus\{c\},\quad
      d_B(f(x),\ell)<g(\varepsilon)\bigr)
    \end{gathered}
  \]
\end{thm}

\begin{proof}
  Similar to the \xref[sequence challenge-replacement proof]{lec04::thm:convergence-challenge-replacement}
\end{proof}

\section{Function Limit Operator}

\begin{dfn}[Limit of a Function at a Point]{function-limit-relation}
  Let \((X,d_X)\) and \((B,d_B)\) be
  \xref[metric spaces]{lec03::def:metric-space}, let \(A\subseteq X\)
  and \(f:A\to B\), and let \(c\in X\) be a
  \xref[limit point]{lec03::def:limit-points} of \(A\).
  We say that \(f\) \emph{has limit \(\ell\in B\) at \(c\)} if
  \[
    \begin{gathered}
      \forall\varepsilon\in\apos{\RR},\quad
      \exists\delta\in\apos{\RR},\quad\forall x\in A,\\
      0<d_X(x,c)<\delta\implies d_B(f(x),\ell)<\varepsilon
    \end{gathered}
  \]
\end{dfn}

\begin{dfn}[Limit Domain]{function-limit-domain}
  A point in the limit domain if and only if
  \begin{enumerate}[nosep]
    \item \(c\) is a \xref[limit point]{lec03::def:limit-points} of the domain \(A\)
    \item \(f\) \xref[has a limit]{def:limit-of-functions} at \(c\)
  \end{enumerate}
  \[
    \LimDom:(A\to B)\to\PowerSet(X)
  \]
  \[
    \LimDom(f)\defeq
    \left\{c\in X\;\middle|\;
      \begin{gathered}
        \forall r\in\apos{\RR},\quad
        \ball{r}{c}\cap\bigl(A\setminus\{c\}\bigr)\neq\emptyset\\
        {}\land{}\\
        \exists\ell\in B,\quad\forall\varepsilon\in\apos{\RR},\quad
        \exists\delta\in\apos{\RR},\\
        \forall x\in A,\quad
        0<d_X(x,c)<\delta\implies d_B(f(x),\ell)<\varepsilon
      \end{gathered}
    \right\}
  \]
  \par\smallskip\noindent{\footnotesize\texttt{LaTeX: \detokenize{\LimDom(f)}}}
\end{dfn}

\begin{prop}[Well-Definedness of the Function Limit]{function-limit-well-defined}
  With the setup of the \xref[limit relation]{def:function-limit-relation},
  for every \(f:A\to B\) and \(c\in\LimDom(f)\), there exists a unique
  \(\ell\in B\) such that \(f\)
  \xref[has limit \(\ell\) at \(c\)]{def:function-limit-relation}
\end{prop}

\begin{proof}
  Existence follows from the \xref[definition of the limit domain]{def:function-limit-domain}.
  The limit is unique by the \xref[uniqueness theorem]{thm:function-limit-uniqueness}
\end{proof}

\begin{dfn}[Function Limit Operator]{function-limit-op}
  \[
    \fnlim:\Pi f:(A\to B),\quad\LimDom(f)\to B
  \]
  % Characterization for f:A->B, c in LimDom(f), and ell in B.
  \[
    \fnlim(f)(c)=\ell
    \iff
    \begin{gathered}
      \forall\varepsilon\in\apos{\RR},\quad
      \exists\delta\in\apos{\RR},\quad\forall x\in A,\\
      0<d_X(x,c)<\delta\implies d_B(f(x),\ell)<\varepsilon
    \end{gathered}
  \]
  \par\smallskip\noindent{\footnotesize\texttt{LaTeX: \detokenize{\fnlim(f)(c)}}}
\end{dfn}

\begin{conv}[Function Limit Notation]{function-limit-notation}
  For the \xref[function limit operator]{def:function-limit-op}, write
  \[
    \fnlim_c(f)\defeq\fnlim(f)(c)
  \]
  \par\smallskip\noindent{\footnotesize\texttt{LaTeX: \detokenize{\fnlim_c(f)}}}
\end{conv} 

\section{Properties of Function Limits} 

\begin{thm}
  If we can propose a convergence sequence $a$ which codomain is in the domain of a function $f$ exculding its $c$, where $\seqlim a = c$, then the limit of the function $f$ at $c$ is equal to the limit of the sequence $n \map f(a(n))$. 
\[f: A \to B, \, \fnlim_c(f) = \ell \iff \forall a: \mathbb{N} \to A \setminus \{c\}, \, \seqlim a = c, \, \seqlim (n \map f(a(n))) = \ell\]
\end{thm}

\section{One-Sided Limits}

\begin{dfn}[Right-Hand Limit]{right-hand-limit}
  Let \(A\subseteq\RR\), let \((B,d_B)\) be a
  \xref[metric space]{lec03::def:metric-space}, and let \(f:A\to B\).
  Let \(c\in\RR\) satisfy
  \[
    \forall r\in\apos{\RR},\quad\exists x\in A,\quad 0<x-c<r
  \]
  We say that \(f\) has \emph{right-hand limit} \(\ell\in B\) at \(c\) if
  \[
    \forall\varepsilon\in\apos{\RR},\,
    \exists\delta\in\apos{\RR},\,\forall x\in A\cap(c,c+\delta),\,
    d_B(f(x),\ell)<\varepsilon
  \]
\end{dfn}

\begin{dfn}[Left-Hand Limit]{left-hand-limit}
  Let \(A\subseteq\RR\), let \((B,d_B)\) be a
  \xref[metric space]{lec03::def:metric-space}, and let \(f:A\to B\).
  Let \(c\in\RR\) satisfy
  \[
    \forall r\in\apos{\RR},\quad\exists x\in A,\quad 0<c-x<r
  \]
  We say that \(f\) has \emph{left-hand limit} \(\ell\in B\) at \(c\) if
  \[
    \forall\varepsilon\in\apos{\RR},\,
    \exists\delta\in\aneg{\RR},\,\forall x\in A\cap(c+\delta,c),\,
    d_B(f(x),\ell)<\varepsilon
  \]
\end{dfn}

\begin{thm}[Uniqueness of One-Sided Limits]{one-sided-limit-uniqueness}
  Under the assumptions of the \xref[right-hand]{def:right-hand-limit}
  or \xref[left-hand]{def:left-hand-limit} definition, the corresponding
  one-sided limit, if it exists, is unique
\end{thm}

\begin{proof}
  The argument is the same as for \xref[two-sided limits]{thm:function-limit-uniqueness},
  choosing \(x\) in the shared one-sided interval using the corresponding
  accumulation assumption
\end{proof}

\begin{dfn}[Right-Hand Limit Domain]{right-hand-limit-domain}
  Let \(A\subseteq\RR\) and let \((B,d_B)\) be a
  \xref[metric space]{lec03::def:metric-space}
  \[
    \LimDomRight:(A\to B)\to\PowerSet(\RR)
  \]
  \[
    \LimDomRight(f)\defeq
    \left\{c\in\RR\;\middle|\;
      \begin{gathered}
        \forall r\in\apos{\RR},\quad A\cap(c,c+r)\neq\emptyset\\
        {}\land{}\\
        \exists\ell\in B,\quad\forall\varepsilon\in\apos{\RR},\quad
        \exists\delta\in\apos{\RR},\\
        \forall x\in A\cap(c,c+\delta),\quad d_B(f(x),\ell)<\varepsilon
      \end{gathered}
    \right\}
  \]
  \par\smallskip\noindent{\footnotesize\texttt{LaTeX: \detokenize{\LimDomRight(f)}}}
\end{dfn}

\begin{dfn}[Left-Hand Limit Domain]{left-hand-limit-domain}
  Let \(A\subseteq\RR\) and let \((B,d_B)\) be a
  \xref[metric space]{lec03::def:metric-space}
  \[
    \LimDomLeft:(A\to B)\to\PowerSet(\RR)
  \]
  \[
    \LimDomLeft(f)\defeq
    \left\{c\in\RR\;\middle|\;
      \begin{gathered}
        \forall r\in\apos{\RR},\quad A\cap(c-r,c)\neq\emptyset\\
        {}\land{}\\
        \exists\ell\in B,\quad\forall\varepsilon\in\apos{\RR},\quad
        \exists\delta\in\aneg{\RR},\\
        \forall x\in A\cap(c+\delta,c),\quad d_B(f(x),\ell)<\varepsilon
      \end{gathered}
    \right\}
  \]
  \par\smallskip\noindent{\footnotesize\texttt{LaTeX: \detokenize{\LimDomLeft(f)}}}
\end{dfn}

\begin{prop}[Well-Definedness of One-Sided Limits]{one-sided-limit-well-defined}
  For every \(f:A\to B\) and \(c\in\LimDomRight(f)\), there is a unique
  \xref[right-hand limit]{def:right-hand-limit} in \(B\).
  The same holds for \(c\in\LimDomLeft(f)\) and the
  \xref[left-hand limit]{def:left-hand-limit}
\end{prop}

\begin{proof}
  Existence follows from the definitions of the
  \xref[right-hand]{def:right-hand-limit-domain} and
  \xref[left-hand]{def:left-hand-limit-domain} limit domains.
  Uniqueness follows from the \xref[one-sided uniqueness theorem]{thm:one-sided-limit-uniqueness}
\end{proof}

\begin{dfn}[Right-Hand Limit Operator]{right-hand-limit-op}
  The operator
  \[
    \rightfnlim:\Pi f:(A\to B),\quad\LimDomRight(f)\to B
  \]
  sends \(f\) and \(c\in\LimDomRight(f)\) to the unique
  \xref[right-hand limit]{def:right-hand-limit} of \(f\) at \(c\)
  \par\smallskip\noindent{\footnotesize\texttt{LaTeX: \detokenize{\rightfnlim(f)(c)}}}
\end{dfn}

\begin{dfn}[Left-Hand Limit Operator]{left-hand-limit-op}
  The operator
  \[
    \leftfnlim:\Pi f:(A\to B),\quad\LimDomLeft(f)\to B
  \]
  sends \(f\) and \(c\in\LimDomLeft(f)\) to the unique
  \xref[left-hand limit]{def:left-hand-limit} of \(f\) at \(c\)
  \par\smallskip\noindent{\footnotesize\texttt{LaTeX: \detokenize{\leftfnlim(f)(c)}}}
\end{dfn}

\begin{conv}[One-Sided Limit Notation]{one-sided-limit-notation}
  For \(c\in\LimDomRight(f)\) or \(c\in\LimDomLeft(f)\), respectively, write
  \[
    \poslim{c}(f)\defeq\rightfnlim(f)(c),
    \qquad
    \neglim{c}(f)\defeq\leftfnlim(f)(c)
  \]
  \par\smallskip\noindent{\footnotesize\texttt{LaTeX: \detokenize{\poslim{3}(f)}, \detokenize{\neglim{3}(f)}}}
\end{conv}

\section{A Systematic Method for Proving Limits}

The shared error-bound method covers finite-value limits of sequences,
two-sided function limits, one-sided limits, and limits as the input tends
to infinity. The input restriction changes with the approach, while the
output tolerance is handled using
\xref[challenge functions]{lec04::def:convergence-challenge-function}.
Infinite output limits use arbitrarily large output thresholds instead

\begin{rem}[Using the Ball Radius as the Parameter]{ball-radius-parameter}
  In the \xref[challenge-function method]{thm:function-limit-challenge-replacement},
  when the input-radius parameter itself supplies the witness \(\delta\),
  the following condition suffices
  \[
    \forall\varepsilon\in\apos{\RR},\quad
    \forall x\in A\cap\ball{\varepsilon}{c}\setminus\{c\},\quad
    d_B(f(x),\ell)\leq g(\varepsilon)
  \]
  Here \(\varepsilon\) is the input-radius parameter, and
  \(g(\varepsilon)\) is the output-error bound for a
  \xref[challenge function]{lec04::def:convergence-challenge-function} \(g\).
  A local restriction such as an input radius at most \(1\) can instead
  be handled by choosing \(\delta=\min(\varepsilon,1)\) and verifying
  the output bound on that smaller ball
\end{rem}
